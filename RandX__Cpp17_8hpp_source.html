173 inline constexpr std::uint64_t RotL(const std::uint64_t x, const int s) noexcept
174 {
175 const int count = s & 63;
176 return count == 0 ? x : ((x << count) | (x >> (64 - count)));

340 // 为数组状态引擎提供公共接口：min/max/discard/serialize/比较/构造/jumpPoly
341 // SplitMix64（标量状态）和 ChaCha20（CSPRNG）不继承此基类

344 template <class Derived, class ResultType, std::size_t N>
345 struct EngineBase
346 {
347 using result_type = ResultType;
348 using state_type = std::array<ResultType, N>;
349
350 // --- 公共接口 ---
351
352 [[nodiscard]]
353 static constexpr result_type min() noexcept
354 {
355 return std::numeric_limits<result_type>::lowest();
356 }
357
358 [[nodiscard]]
359 static constexpr result_type max() noexcept
360 {
361 return std::numeric_limits<result_type>::max();
362 }
363
364 constexpr void discard(unsigned long long z) noexcept
365 {
366 for (unsigned long long i = 0; i < z; ++i)
367 static_cast<Derived*>(this)->operator()();
368 }
369
370 [[nodiscard]]
371 constexpr state_type serialize() const noexcept
372 {
373 return s_;
374 }
375
376 constexpr void deserialize(const state_type& s) noexcept
377 {
378 s_ = s;
379 if constexpr (!EngineStatePolicy<Derived>::AllowsZeroState)
380 {
381 if (IsAllZero(s_))
382 {
383 s_[0] = static_cast<ResultType>(1);
384 }
385 assert(!IsAllZero(s_) && "absorbing all-zero state");
386 }
387 }
388
389 // C++17: 手写比较运算符
390 friend bool operator==(const EngineBase& lhs, const EngineBase& rhs) noexcept
391 {
392 return lhs.s_ == rhs.s_;
393 }
394
395 friend bool operator!=(const EngineBase& lhs, const EngineBase& rhs) noexcept
396 {
397 return lhs.s_ != rhs.s_;
398 }
399
400 protected:
401
402 static constexpr int Bits = static_cast<int>(sizeof(ResultType) * 8);
403
404 EngineBase() = default;
405
406 // State 构造（用户直接传入，包含 Release/Debug 全零状态静默修正）
407 explicit constexpr EngineBase(const state_type& state) noexcept
408 : s_(state)
409 {
410 if constexpr (!EngineStatePolicy<Derived>::AllowsZeroState)
411 {
412 if (IsAllZero(s_))
413 {
414 s_[0] = static_cast<ResultType>(1);
415 }
416 assert(!IsAllZero(s_) && "absorbing all-zero state");
417 }
418 }
419
420 // SeedSeq 构造（SFINAE 排除 state_type 和 Derived）
421 template <class SeedSeq,
422 std::enable_if_t<detail::is_seed_sequence_v<SeedSeq>
423 && !std::is_same_v<std::decay_t<SeedSeq>, state_type>
424 && !std::is_same_v<std::decay_t<SeedSeq>, Derived>>* = nullptr>
425 explicit constexpr EngineBase(SeedSeq& seq)
426 {
427 if constexpr (sizeof(result_type) == 8)
428 {
429 std::array<std::uint32_t, N * 2> raw;
430 seq.generate(raw.data(), raw.data() + raw.size());
431 for (std::size_t i = 0; i < N; ++i)
432 s_[i] = (static_cast<result_type>(raw[2 * i]) << 32) | raw[2 * i + 1];
433 }
434 else
435 {
436 std::array<std::uint32_t, N> raw;
437 seq.generate(raw.data(), raw.data() + raw.size());
438 for (std::size_t i = 0; i < N; ++i)
439 s_[i] = static_cast<result_type>(raw[i]);
440 }
441 if (IsAllZero(s_)) s_[0] = 1;
442 }
443
444 // 单值播种（SplitMix64 扩展，等价于 generateSeedSequence<N>）
445 explicit constexpr EngineBase(std::uint64_t seed) noexcept
446 {
447 SplitMix64 sm{ seed };
448 for (std::size_t i = 0; i < N; ++i)
449 s_[i] = static_cast<result_type>(sm());
450 if (IsAllZero(s_)) s_[0] = 1;
451 }
452
453 // jump 多项式通用实现（constexpr，供 MakeStreamEngine 编译期调用）
454 template <std::size_t K>
455 constexpr void jumpPoly(const ResultType (&poly)[K]) noexcept
456 {
457 std::array<ResultType, N> acc{};
458 for (std::size_t i = 0; i < K; ++i)
459 for (int b = 0; b < Bits; ++b)
460 {
461 if (poly[i] & (ResultType{ 1 } << b))
462 for (std::size_t j = 0; j < N; ++j)
463 acc[j] ^= s_[j];
464 static_cast<Derived*>(this)->operator()();
465 }
466 s_ = acc;
467 }
468
469 state_type s_{};
470 };
471 }

949 // ── A3 跨平台 OS 密码学熵源 ──
950 // 用 OS 密码学 API 填充 [buf, buf+n) 字节；成功返回 true，失败或不支持返回 false。
951 // 平台支持：Windows BCryptGenRandom、Linux getrandom、macOS SecRandomCopyBytes。
952 // 注：密码学安全组件在 OS 熵源不可用或失败时直接抛异常，绝不隐式降级至 std::random_device 或非安全源；
953 // getrandom 可能短读，内部循环直至填满；BCryptGenRandom/SecRandomCopyBytes 一次填满。
954 [[nodiscard]]
955 inline bool GetOsEntropyBytes(void* buf, std::size_t n) noexcept
956 {
957 if (n == 0) return true;
958 auto* p = static_cast<std::uint8_t*>(buf);
959
960# if defined(_WIN32) && __has_include(<bcrypt.h>)
961 // Windows: BCryptGenRandom（分块处理 >4GB 时的 ULONG 截断）
962 // NTSTATUS >= 0 即 NT_SUCCESS（使用 BCRYPT_SUCCESS 宏或强转 NTSTATUS 判定）
963 std::size_t filled = 0;
964 while (filled < n)
965 {
966 const ULONG chunkSize = static_cast<ULONG>((std::min)(n - filled, static_cast<std::size_t>((std::numeric_limits<ULONG>::max)())));
967 const auto status = ::BCryptGenRandom(nullptr, p + filled, chunkSize, BCRYPT_USE_SYSTEM_PREFERRED_RNG);
968# if defined(BCRYPT_SUCCESS)
969 if (!BCRYPT_SUCCESS(status)) return false;
970# else
971 if (static_cast<NTSTATUS>(status) < 0) return false;
972# endif
973 filled += chunkSize;
974 }
975 return true;
976
977# elif defined(__linux__) && __has_include(<sys/random.h>)
978 // Linux: getrandom（循环处理短读与 EINTR）
979 std::size_t filled = 0;
980 while (filled < n)
981 {
982 const ssize_t ret = ::getrandom(p + filled, n - filled, 0);
983 if (ret < 0)
984 {
985 if (errno == EINTR) continue; // 被信号打断，重试
986 return false; // ENOSYS/EFAULT 等不可恢复错误
987 }
988 if (ret == 0) return false;
989 filled += static_cast<std::size_t>(ret);
990 }
991 return true;
992
993# elif defined(__APPLE__) && __has_include(<Security/Security.h>)
994 // macOS: SecRandomCopyBytes（一次调用填满）
995 return (::SecRandomCopyBytes(kSecRandomDefault, n, p) == errSecSuccess);
996
997# else
998 // 无可用 OS 密码学熵源 → 返回 false，SecureRandomBytes 将抛出异常
999 // 非安全场景的播种请使用 RandomSeed()（含 random_device → 时间戳回退链）
1000 (void)p; (void)n;
1001 return false;
1002# endif
1003 }

1753 // 生成一个 ChaCha20 block（64 字节）填充 m_buffer
1754 inline void ChaCha20::generateBlock()
1755 {
1756 if (m_counterExhausted)
1757 {
1758 throw std::overflow_error("ChaCha20: 32-bit block counter overflow");
1759 }
1760
1761 // 构造完整 16-word 状态：常数 + key + counter + nonce
1762 std::array<std::uint32_t, 16> state{};
1763 detail::ScopedWiper stateWiper(state.data(), sizeof(state));
1764 state[0] = detail::ChaCha20Constants[0];
1765 state[1] = detail::ChaCha20Constants[1];
1766 state[2] = detail::ChaCha20Constants[2];
1767 state[3] = detail::ChaCha20Constants[3];
1768 for (int i = 0; i < 8; ++i) state[4 + i] = m_state[i]; // key
1769 state[12] = m_state[8]; // counter
1770 state[13] = m_state[9]; // nonce[0]
1771 state[14] = m_state[10]; // nonce[1]
1772 state[15] = m_state[11]; // nonce[2]
1773
1774 std::array<std::uint32_t, 16> working = state;
1775 detail::ScopedWiper workingWiper(working.data(), sizeof(working));
1776
1777 // 20 轮 = 10 次 double-round（列轮 + 对角轮）
1778 for (int i = 0; i < 10; ++i)
1779 {
1780 // 列轮 QR 顺序：(0,4,8,12) (1,5,9,13) (2,6,10,14) (3,7,11,15)
1781 detail::ChaCha20QuarterRound(working[0], working[4], working[8], working[12]);
1782 detail::ChaCha20QuarterRound(working[1], working[5], working[9], working[13]);
1783 detail::ChaCha20QuarterRound(working[2], working[6], working[10], working[14]);
1784 detail::ChaCha20QuarterRound(working[3], working[7], working[11], working[15]);
1785 // 对角轮 QR 顺序：(0,5,10,15) (1,6,11,12) (2,7,8,13) (3,4,9,14)
1786 detail::ChaCha20QuarterRound(working[0], working[5], working[10], working[15]);
1787 detail::ChaCha20QuarterRound(working[1], working[6], working[11], working[12]);
1788 detail::ChaCha20QuarterRound(working[2], working[7], working[8], working[13]);
1789 detail::ChaCha20QuarterRound(working[3], working[4], working[9], working[14]);
1790 }
1791
1792 // 加初始状态后按小端序输出 64 字节到 m_buffer
1793 for (int i = 0; i < 16; ++i)
1794 {
1795 const std::uint32_t v = working[i] + state[i];
1796 m_buffer[i * 4 + 0] = static_cast<std::uint8_t>(v);
1797 m_buffer[i * 4 + 1] = static_cast<std::uint8_t>(v >> 8);
1798 m_buffer[i * 4 + 2] = static_cast<std::uint8_t>(v >> 16);
1799 m_buffer[i * 4 + 3] = static_cast<std::uint8_t>(v >> 24);
1800 }
1801
1802 if (m_state[8] == 0xFFFFFFFFU)
1803 {
1804 m_counterExhausted = true;
1805 }
1806 else
1807 {
1808 ++m_state[8];
1809 }
1810 m_bufferPos = 0;
1811 }

2747 // 路径 1：随机访问迭代器（hash-set / 索引数组双分支）
2748 template <class It,
2749 std::enable_if_t<detail::is_random_access_iterator_v<It>>* = nullptr>
2750 [[nodiscard]]
2751 inline std::vector<typename std::iterator_traits<It>::value_type>
2752 RandSample(It first, It last, typename std::iterator_traits<It>::difference_type n)
2753 {
2754 using Diff = typename std::iterator_traits<It>::difference_type;
2755 using T = typename std::iterator_traits<It>::value_type;
2756 const Diff size = std::distance(first, last);
2757 if (n <= 0 || size <= 0)
2758 return {};
2759 if (n >= size)
2760 {
2761 std::vector<T> all;
2762 all.reserve(static_cast<std::size_t>(size));
2763 for (Diff i = 0; i < size; ++i)
2764 all.push_back(first[i]);
2765 return all;
2766 }
2767
2768 auto& rng = DefaultEngine();
2769
2770 const auto sizeU = static_cast<std::uint64_t>(size);
2771 const auto nU = static_cast<std::uint64_t>(n);
2772 const auto nSample = static_cast<std::size_t>(n);
2773
2774 // 分支选择：n·K < size 时 hash-set 内存优（O(n)）；否则索引数组常数优（O(N)）
2775 if (nU <= (sizeU - 1) / detail::HashSetThresholdK)
2776 {
2777 // hash-set 分支：O(n) 内存，O(n) 期望时间
2778 std::unordered_set<std::uint64_t> selected;
2779 selected.reserve(nSample);
2780 std::vector<T> result;
2781 result.reserve(nSample);
2782 std::uniform_int_distribution<std::uint64_t> dist(0, sizeU - 1);
2783 while (result.size() < nSample)
2784 {
2785 const std::uint64_t idx = dist(rng);
2786 if (selected.insert(idx).second)
2787 result.push_back(first[static_cast<Diff>(idx)]);
2788 }
2789 return result;
2790 }
2791
2792 // 索引数组分支：O(N) 内存，O(N) 时间，无碰撞
2793 const std::size_t sz = static_cast<std::size_t>(size);
2794 std::vector<std::size_t> indices(sz);
2795 for (std::size_t i = 0; i < sz; ++i)
2796 indices[i] = i;
2797
2798 // Fisher-Yates 前 n 步：j ∈ [i, size-1]
2799 for (std::size_t i = 0; i < nSample; ++i)
2800 {
2801 std::uniform_int_distribution<std::size_t> dist(i, sz - 1);
2802 const std::size_t j = dist(rng);
2803 std::swap(indices[i], indices[j]);
2804 }
2805
2806 std::vector<T> result;
2807 result.reserve(nSample);
2808 for (std::size_t i = 0; i < nSample; ++i)
2809 result.push_back(first[static_cast<Diff>(indices[i])]);
2810 return result;
2811 }

2813 // 路径 2：输入迭代器（reservoir sampling, Algorithm R）
2814 template <class It,
2815 std::enable_if_t<detail::is_input_iterator_v<It>

2817 [[nodiscard]]
2818 inline std::vector<typename std::iterator_traits<It>::value_type>
2819 RandSample(It first, It last, typename std::iterator_traits<It>::difference_type n)
2820 {
2821 using Diff = typename std::iterator_traits<It>::difference_type;
2822 using T = typename std::iterator_traits<It>::value_type;
2823 if (n <= 0)
2824 return {};
2825
2826 std::vector<T> reservoir;
2827 reservoir.reserve(static_cast<std::size_t>(n));
2828
2829 // 填满蓄水池
2830 Diff i = 0;
2831 for (; i < n && first != last; ++i, ++first)

2840 for (; first != last; ++i, ++first)
2841 {

2843 const auto j = dist(rng);
2844 if (j < static_cast<std::uint64_t>(n))
2845 reservoir[static_cast<std::size_t>(j)] = *first;
2846 }
