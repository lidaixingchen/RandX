336 // 为数组状态引擎提供公共接口：min/max/discard/serialize/比较/构造/jumpPoly

340 template <class Derived, class ResultType, std::size_t N>
341 struct EngineBase
342 {
343 using result_type = ResultType;
344 using state_type = std::array<ResultType, N>;
345
346 // --- 公共接口 ---
347
348 [[nodiscard]]
349 static constexpr result_type min() noexcept
350 {
351 return std::numeric_limits<result_type>::lowest();
352 }
353
354 [[nodiscard]]
355 static constexpr result_type max() noexcept
356 {
357 return std::numeric_limits<result_type>::max();
358 }
359
360 constexpr void discard(unsigned long long z) noexcept
361 {
362 for (unsigned long long i = 0; i < z; ++i)
363 static_cast<Derived*>(this)->operator()();
364 }
365
366 [[nodiscard]]
367 constexpr state_type serialize() const noexcept
368 {
369 return s_;
370 }
371
372 constexpr void deserialize(const state_type& s) noexcept
373 {
374 s_ = s;
375 if constexpr (!EngineStatePolicy<Derived>::AllowsZeroState)
376 {
377 if (IsAllZero(s_))
378 {
379 s_[0] = static_cast<ResultType>(1);
380 }
381 assert(!IsAllZero(s_) && "absorbing all-zero state");
382 }
383 }
384
385 // C++23: defaulted 三路比较（保留 ==, !=, <, >, <=, >= 全套）
386 friend auto operator<=>(const EngineBase&, const EngineBase&) = default;
387
388 protected:
389
390 static constexpr int Bits = static_cast<int>(sizeof(ResultType) * 8);
391
392 EngineBase() = default;
393
394 // State 构造（用户直接传入，包含 Release/Debug 全零状态静默修正）
395 explicit constexpr EngineBase(const state_type& state) noexcept
396 : s_(state)
397 {
398 if constexpr (!EngineStatePolicy<Derived>::AllowsZeroState)
399 {
400 if (IsAllZero(s_))
401 {
402 s_[0] = static_cast<ResultType>(1);
403 }
404 assert(!IsAllZero(s_) && "absorbing all-zero state");
405 }
406 }
407
408 // SeedSeq 构造（零状态修正，Release 安全）
409 template <detail::SeedSequence SeedSeq>
410 requires (!std::same_as<std::remove_cvref_t<SeedSeq>, state_type>
411 && !std::same_as<std::remove_cvref_t<SeedSeq>, Derived>)
412 explicit constexpr EngineBase(SeedSeq& seq)
413 {
414 if constexpr (sizeof(result_type) == 8)
415 {
416 std::array<std::uint32_t, N * 2> raw;
417 seq.generate(raw.data(), raw.data() + raw.size());
418 for (std::size_t i = 0; i < N; ++i)
419 s_[i] = (static_cast<result_type>(raw[2 * i]) << 32) | raw[2 * i + 1];
420 }
421 else
422 {
423 std::array<std::uint32_t, N> raw;
424 seq.generate(raw.data(), raw.data() + raw.size());
425 for (std::size_t i = 0; i < N; ++i)
426 s_[i] = static_cast<result_type>(raw[i]);
427 }
428 if (IsAllZero(s_)) s_[0] = 1;
429 }
430
431 // 单值播种（SplitMix64 扩展，等价于 generateSeedSequence<N>）
432 explicit constexpr EngineBase(std::uint64_t seed) noexcept
433 {
434 SplitMix64 sm{ seed };
435 for (std::size_t i = 0; i < N; ++i)
436 s_[i] = static_cast<result_type>(sm());
437 if (IsAllZero(s_)) s_[0] = 1;
438 }
439
440 // jump 多项式通用实现（constexpr，供 MakeStreamEngine 编译期调用）
441 template <std::size_t K>
442 constexpr void jumpPoly(const ResultType (&poly)[K]) noexcept
443 {
444 std::array<ResultType, N> acc{};
445 for (std::size_t i = 0; i < K; ++i)
446 for (int b = 0; b < Bits; ++b)
447 {
448 if (poly[i] & (ResultType{ 1 } << b))
449 for (std::size_t j = 0; j < N; ++j)
450 acc[j] ^= s_[j];
451 static_cast<Derived*>(this)->operator()();
452 }
453 s_ = acc;
454 }
455
456 state_type s_{};
457 };
458 }

1259 template <class T>
1260 inline T NormalizeBetaSample(T x, T y)
1261 {
1262 if (!std::isfinite(x) || !std::isfinite(y) || x < T{0} || y < T{0})
1263 throw std::domain_error("RandBeta: Gamma sample is not finite or negative");
1264 if (x == T{0} && y == T{0})
1265 throw std::domain_error("RandBeta: both Gamma samples are zero");
1266 if (x == T{0}) return T{0};
1267 if (y == T{0}) return T{1};
1268 const T sum = x + y;
1269 if (std::isfinite(sum))
1270 {
1271 return x / sum;
1272 }
1273 if (x >= y && x > T{0})
1274 {
1275 return T{1} / (T{1} + (y / x));
1276 }
1277 else if (x < y && y > T{0})
1278 {
1279 const T r = x / y;
1280 return r / (T{1} + r);
1281 }
1282 throw std::domain_error("RandBeta: unable to normalize Gamma samples");
1283 }

1941 // 生成一个 ChaCha20 block（64 字节）填充 m_buffer
1942 inline void ChaCha20::generateBlock()
1943 {
1944 if (m_counterExhausted)
1945 {
1946 throw std::overflow_error("ChaCha20: 32-bit block counter overflow");
1947 }
1948
1949 // 构造完整 16-word 状态：常数 + key + counter + nonce
1950 std::array<std::uint32_t, 16> state{};
1951 detail::ScopedWiper stateWiper(state.data(), sizeof(state));
1952 state[0] = detail::ChaCha20Constants[0];
1953 state[1] = detail::ChaCha20Constants[1];
1954 state[2] = detail::ChaCha20Constants[2];
1955 state[3] = detail::ChaCha20Constants[3];
1956 for (int i = 0; i < 8; ++i) state[4 + i] = m_state[i]; // key
1957 state[12] = m_state[8]; // counter
1958 state[13] = m_state[9]; // nonce[0]
1959 state[14] = m_state[10]; // nonce[1]
1960 state[15] = m_state[11]; // nonce[2]
1961
1962 std::array<std::uint32_t, 16> working = state;
1963 detail::ScopedWiper workingWiper(working.data(), sizeof(working));
1964
1965 // 20 轮 = 10 次 double-round（列轮 + 对角轮）
1966 for (int i = 0; i < 10; ++i)
1967 {
1968 // 列轮 QR 顺序：(0,4,8,12) (1,5,9,13) (2,6,10,14) (3,7,11,15)
1969 detail::ChaCha20QuarterRound(working[0], working[4], working[8], working[12]);
1970 detail::ChaCha20QuarterRound(working[1], working[5], working[9], working[13]);
1971 detail::ChaCha20QuarterRound(working[2], working[6], working[10], working[14]);
1972 detail::ChaCha20QuarterRound(working[3], working[7], working[11], working[15]);
1973 // 对角轮 QR 顺序：(0,5,10,15) (1,6,11,12) (2,7,8,13) (3,4,9,14)
1974 detail::ChaCha20QuarterRound(working[0], working[5], working[10], working[15]);
1975 detail::ChaCha20QuarterRound(working[1], working[6], working[11], working[12]);
1976 detail::ChaCha20QuarterRound(working[2], working[7], working[8], working[13]);
1977 detail::ChaCha20QuarterRound(working[3], working[4], working[9], working[14]);

4443 {

4445 WorkT delta = WorkT{0};
4446
4447 if (a_sample.has_extra && b_sample.has_extra)
4448 {
4449 const WorkT sa = a_sample.shape;

4451 if (sa == sb)
4452 {
4453 const WorkT diff = a_sample.exp_val - b_sample.exp_val;
4454 delta = diff / sa;
4455 }
4456 else if (sa < sb)
4457 {
4458 const WorkT r = sa / sb;
4459 const WorkT diff = a_sample.exp_val - r * b_sample.exp_val;

4461 }
4462 else
4463 {
4464 const WorkT r = sb / sa;
4465 const WorkT diff = r * a_sample.exp_val - b_sample.exp_val;
4466 delta = diff / sb;
4467 }
4468 }

4470 {

4474 {
4475 delta = -b_sample.exp_val / b_sample.shape;
4476 }

4482 throw std::runtime_error("RandBeta: NaN encountered during Beta calculation");

4486 {
4487 const WorkT em1 = std::expm1(L);
4488 return WorkT{0.5} - em1 / (WorkT{2} * (WorkT{2} + em1));
