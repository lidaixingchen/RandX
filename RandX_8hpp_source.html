331 // 为数组状态引擎提供公共接口：min/max/discard/serialize/比较/构造/jumpPoly
332 // SplitMix64（标量状态）和 ChaCha20（CSPRNG）不继承此基类

335 template <class Derived, class ResultType, std::size_t N>
336 struct EngineBase
337 {
338 using result_type = ResultType;
339 using state_type = std::array<ResultType, N>;
340
341 // --- 公共接口 ---
342
343 [[nodiscard]]
344 static constexpr result_type min() noexcept
345 {
346 return std::numeric_limits<result_type>::lowest();
347 }
348
349 [[nodiscard]]
350 static constexpr result_type max() noexcept
351 {
352 return std::numeric_limits<result_type>::max();
353 }
354
355 constexpr void discard(unsigned long long z) noexcept
356 {
357 for (unsigned long long i = 0; i < z; ++i)
358 static_cast<Derived*>(this)->operator()();
359 }
360
361 [[nodiscard]]
362 constexpr state_type serialize() const noexcept
363 {
364 return s_;
365 }
366
367 constexpr void deserialize(const state_type& s) noexcept
368 {
369 s_ = s;
370 if constexpr (!EngineStatePolicy<Derived>::AllowsZeroState)
371 {
372 if (IsAllZero(s_))
373 {
374 s_[0] = static_cast<ResultType>(1);
375 }
376 assert(!IsAllZero(s_) && "absorbing all-zero state");
377 }
378 }
379
380 // C++23: defaulted 三路比较（保留 ==, !=, <, >, <=, >= 全套）
381 friend auto operator<=>(const EngineBase&, const EngineBase&) = default;
382
383 protected:
384
385 static constexpr int Bits = static_cast<int>(sizeof(ResultType) * 8);
386
387 EngineBase() = default;
388
389 // State 构造（用户直接传入，包含 Release/Debug 全零状态静默修正）
390 explicit constexpr EngineBase(const state_type& state) noexcept
391 : s_(state)
392 {
393 if constexpr (!EngineStatePolicy<Derived>::AllowsZeroState)
394 {
395 if (IsAllZero(s_))
396 {
397 s_[0] = static_cast<ResultType>(1);
398 }
399 assert(!IsAllZero(s_) && "absorbing all-zero state");
400 }
401 }
402
403 // SeedSeq 构造（零状态修正，Release 安全）
404 template <detail::SeedSequence SeedSeq>
405 requires (!std::same_as<std::remove_cvref_t<SeedSeq>, state_type>
406 && !std::same_as<std::remove_cvref_t<SeedSeq>, Derived>)
407 explicit constexpr EngineBase(SeedSeq& seq)
408 {
409 if constexpr (sizeof(result_type) == 8)
410 {
411 std::array<std::uint32_t, N * 2> raw;
412 seq.generate(raw.data(), raw.data() + raw.size());
413 for (std::size_t i = 0; i < N; ++i)
414 s_[i] = (static_cast<result_type>(raw[2 * i]) << 32) | raw[2 * i + 1];
415 }
416 else
417 {
418 std::array<std::uint32_t, N> raw;
419 seq.generate(raw.data(), raw.data() + raw.size());
420 for (std::size_t i = 0; i < N; ++i)
421 s_[i] = static_cast<result_type>(raw[i]);
422 }
423 if (IsAllZero(s_)) s_[0] = 1;
424 }
425
426 // 单值播种（SplitMix64 扩展，等价于 generateSeedSequence<N>）
427 explicit constexpr EngineBase(std::uint64_t seed) noexcept
428 {
429 SplitMix64 sm{ seed };
430 for (std::size_t i = 0; i < N; ++i)
431 s_[i] = static_cast<result_type>(sm());
432 if (IsAllZero(s_)) s_[0] = 1;
433 }
434
435 // jump 多项式通用实现（constexpr，供 MakeStreamEngine 编译期调用）
436 template <std::size_t K>
437 constexpr void jumpPoly(const ResultType (&poly)[K]) noexcept
438 {
439 std::array<ResultType, N> acc{};
440 for (std::size_t i = 0; i < K; ++i)
441 for (int b = 0; b < Bits; ++b)
442 {
443 if (poly[i] & (ResultType{ 1 } << b))
444 for (std::size_t j = 0; j < N; ++j)
445 acc[j] ^= s_[j];
446 static_cast<Derived*>(this)->operator()();
447 }
448 s_ = acc;
449 }
450
451 state_type s_{};
452 };
453 }

939 // ── A3 跨平台 OS 密码学熵源 ──
940 // 用 OS 密码学 API 填充 [buf, buf+n) 字节；成功返回 true，失败或不支持返回 false。
941 // 平台支持：Windows BCryptGenRandom、Linux getrandom、macOS SecRandomCopyBytes。
942 // 注：密码学安全组件在 OS 熵源不可用或失败时直接抛异常，绝不隐式降级至 std::random_device 或非安全源；
943 // getrandom 可能短读，内部循环直至填满；BCryptGenRandom/SecRandomCopyBytes 一次填满。
944 [[nodiscard]]
945 inline bool GetOsEntropyBytes(void* buf, std::size_t n) noexcept
946 {
947 if (n == 0) return true;
948 auto* p = static_cast<std::uint8_t*>(buf);
949
950# if defined(_WIN32) && __has_include(<bcrypt.h>)
951 // Windows: BCryptGenRandom（分块处理 >4GB 时的 ULONG 截断）
952 // NTSTATUS >= 0 即 NT_SUCCESS（不能 == 0，正向 informational code 也属成功）
953 std::size_t filled = 0;
954 while (filled < n)
955 {
956 const ULONG chunkSize = static_cast<ULONG>((std::min)(n - filled, static_cast<std::size_t>((std::numeric_limits<ULONG>::max)())));
957 if (::BCryptGenRandom(nullptr, p + filled, chunkSize, BCRYPT_USE_SYSTEM_PREFERRED_RNG) < 0)
958 {
959 return false;
960 }
961 filled += chunkSize;
962 }
963 return true;
964
965# elif defined(__linux__) && __has_include(<sys/random.h>)
966 // Linux: getrandom（循环处理短读与 EINTR）
967 std::size_t filled = 0;
968 while (filled < n)
969 {
970 const ssize_t ret = ::getrandom(p + filled, n - filled, 0);
971 if (ret < 0)
972 {
973 if (errno == EINTR) continue; // 被信号打断，重试
974 return false; // ENOSYS/EFAULT 等不可恢复错误
975 }
976 if (ret == 0) return false;
977 filled += static_cast<std::size_t>(ret);
978 }
979 return true;
980
981# elif defined(__APPLE__) && __has_include(<Security/Security.h>)
982 // macOS: SecRandomCopyBytes（一次调用填满）
983 return (::SecRandomCopyBytes(kSecRandomDefault, n, p) == errSecSuccess);
984
985# else
986 // 无可用 OS 密码学熵源 → 返回 false，SecureRandomBytes 将抛出异常
987 // 非安全场景的播种请使用 RandomSeed()（含 random_device → 时间戳回退链）
988 (void)p; (void)n;
989 return false;
990# endif
991 }

1053 template <class T>
1054 inline T NormalizeBetaSample(T x, T y)
1055 {
1056 if (!std::isfinite(x) || !std::isfinite(y) || x < T{0} || y < T{0})
1057 throw std::domain_error("RandBeta: Gamma sample is not finite or negative");
1058 if (x == T{0} && y == T{0})
1059 throw std::domain_error("RandBeta: both Gamma samples are zero");
1060 if (x == T{0}) return T{0};
1061 if (y == T{0}) return T{1};
1062 const T sum = x + y;
1063 if (std::isfinite(sum))
1064 {
1065 return x / sum;
1066 }
1067 if (x >= y && x > T{0})
1068 {
1069 return T{1} / (T{1} + (y / x));
1070 }
1071 else if (x < y && y > T{0})
1072 {
1073 const T r = x / y;
1074 return r / (T{1} + r);
1075 }
1076 throw std::domain_error("RandBeta: unable to normalize Gamma samples");
1077 }

1676 // 生成一个 ChaCha20 block（64 字节）填充 m_buffer
1677 inline void ChaCha20::generateBlock()
1678 {
1679 if (m_counterExhausted)
1680 {
1681 throw std::overflow_error("ChaCha20: 32-bit block counter overflow");
1682 }
1683
1684 // 构造完整 16-word 状态：常数 + key + counter + nonce
1685 std::array<std::uint32_t, 16> state{};
1686 detail::ScopedWiper stateWiper(state.data(), sizeof(state));
1687 state[0] = detail::ChaCha20Constants[0];
1688 state[1] = detail::ChaCha20Constants[1];
1689 state[2] = detail::ChaCha20Constants[2];
1690 state[3] = detail::ChaCha20Constants[3];
1691 for (int i = 0; i < 8; ++i) state[4 + i] = m_state[i]; // key
1692 state[12] = m_state[8]; // counter
1693 state[13] = m_state[9]; // nonce[0]
1694 state[14] = m_state[10]; // nonce[1]
1695 state[15] = m_state[11]; // nonce[2]
1696
1697 std::array<std::uint32_t, 16> working = state;
1698 detail::ScopedWiper workingWiper(working.data(), sizeof(working));
1699
1700 // 20 轮 = 10 次 double-round（列轮 + 对角轮）
1701 for (int i = 0; i < 10; ++i)
1702 {
1703 // 列轮 QR 顺序：(0,4,8,12) (1,5,9,13) (2,6,10,14) (3,7,11,15)
1704 detail::ChaCha20QuarterRound(working[0], working[4], working[8], working[12]);
1705 detail::ChaCha20QuarterRound(working[1], working[5], working[9], working[13]);
1706 detail::ChaCha20QuarterRound(working[2], working[6], working[10], working[14]);
1707 detail::ChaCha20QuarterRound(working[3], working[7], working[11], working[15]);
1708 // 对角轮 QR 顺序：(0,5,10,15) (1,6,11,12) (2,7,8,13) (3,4,9,14)
1709 detail::ChaCha20QuarterRound(working[0], working[5], working[10], working[15]);
1710 detail::ChaCha20QuarterRound(working[1], working[6], working[11], working[12]);
1711 detail::ChaCha20QuarterRound(working[2], working[7], working[8], working[13]);
1712 detail::ChaCha20QuarterRound(working[3], working[4], working[9], working[14]);
1713 }
1714
1715 // 加初始状态后按小端序输出 64 字节到 m_buffer
1716 for (int i = 0; i < 16; ++i)
1717 {
1718 const std::uint32_t v = working[i] + state[i];
1719 m_buffer[i * 4 + 0] = static_cast<std::uint8_t>(v);
1720 m_buffer[i * 4 + 1] = static_cast<std::uint8_t>(v >> 8);
1721 m_buffer[i * 4 + 2] = static_cast<std::uint8_t>(v >> 16);
1722 m_buffer[i * 4 + 3] = static_cast<std::uint8_t>(v >> 24);
1723 }
1724
1725 if (m_state[8] == 0xFFFFFFFFU)
1726 {
1727 m_counterExhausted = true;
1728 }
1729 else
1730 {
1731 ++m_state[8];
1732 }
1733 m_bufferPos = 0;
1734 }

2670 // 路径 1：随机访问迭代器（hash-set / 索引数组双分支）
2671 template <std::random_access_iterator It, std::sentinel_for<It> Sentinel>
2672 [[nodiscard]]
2673 inline std::vector<std::iter_value_t<It>>
2674 RandSample(It first, Sentinel last, std::iter_difference_t<It> n)
2675 {
2676 using Diff = std::iter_difference_t<It>;
2677 using T = std::iter_value_t<It>;
2678 const Diff size = static_cast<Diff>(std::ranges::distance(first, last));
2679 if (n <= 0 || size <= 0)
2680 return {};
2681 if (n >= size)
2682 {
2683 std::vector<T> all;
2684 all.reserve(static_cast<std::size_t>(size));
2685 for (Diff i = 0; i < size; ++i)
2686 all.push_back(first[i]);
2687 return all;
2688 }
2689
2690 auto& rng = DefaultEngine();
2691
2692 const auto sizeU = static_cast<std::uint64_t>(size);
2693 const auto nU = static_cast<std::uint64_t>(n);
2694 const auto nSample = static_cast<std::size_t>(n);
2695
2696 // 分支选择：n·K < size 时 hash-set 内存优（O(n)）；否则索引数组常数优（O(N)）
2697 if (nU <= (sizeU - 1) / detail::HashSetThresholdK)
2698 {
2699 // hash-set 分支：O(n) 内存，O(n) 期望时间
2700 std::unordered_set<std::uint64_t> selected;
2701 selected.reserve(nSample);
2702 std::vector<T> result;
2703 result.reserve(nSample);
2704 std::uniform_int_distribution<std::uint64_t> dist(0, sizeU - 1);
2705 while (result.size() < nSample)
2706 {
2707 const std::uint64_t idx = dist(rng);
2708 if (selected.insert(idx).second)
2709 result.push_back(first[static_cast<Diff>(idx)]);
2710 }
2711 return result;
2712 }
2713
2714 // 索引数组分支：O(N) 内存，O(N) 时间，无碰撞
2715 const std::size_t sz = static_cast<std::size_t>(size);
2716 std::vector<std::size_t> indices(sz);
2717 for (std::size_t i = 0; i < sz; ++i)
2718 indices[i] = i;
2719
2720 // Fisher-Yates 前 n 步：j ∈ [i, size-1]
2721 for (std::size_t i = 0; i < nSample; ++i)
2722 {
2723 std::uniform_int_distribution<std::size_t> dist(i, sz - 1);
2724 const std::size_t j = dist(rng);
2725 std::swap(indices[i], indices[j]);
2726 }
2727
2728 std::vector<T> result;
2729 result.reserve(nSample);
2730 for (std::size_t i = 0; i < nSample; ++i)
2731 result.push_back(first[static_cast<Diff>(indices[i])]);
2732 return result;
2733 }
